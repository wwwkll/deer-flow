# 智能体收藏功能需求文档

> 目标：为智能体添加收藏功能，支持用户收藏常用智能体，并在左侧菜单快速访问

---

## 1. 需求概述

### 1.1 功能目标

- 在智能体列表页面（Agent Gallery）的每个智能体卡片上添加收藏按钮
- 收藏状态持久化存储到 SQLite 数据库
- 左侧菜单的"智能体"一级菜单下增加二级菜单，显示所有已收藏的智能体
- 点击收藏的智能体可直接进入与该智能体的新对话

### 1.2 业务规则

1. **收藏操作**：用户可在智能体列表页面点击收藏按钮 toggle 收藏状态
2. **收藏存储**：收藏数据存储在 SQLite 数据库的新表 `agent_favorites` 中
3. **左侧菜单展示**：
   - "智能体"一级菜单下增加二级菜单"收藏的智能体"
   - 显示所有已收藏的智能体名称
   - 无收藏时显示空状态提示
4. **快速访问**：点击收藏的智能体直接跳转到 `/workspace/agents/{agent_name}/chats/new`
5. **实时更新**：收藏/取消收藏后，左侧菜单实时更新

## 3. 数据模型设计

### 3.1 数据库表结构

在现有的 `global_variables.db` 数据库中新增 `agent_favorites` 表：

```sql
CREATE TABLE IF NOT EXISTS agent_favorites (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    agent_name TEXT NOT NULL,
    created_at TEXT NOT NULL,
    UNIQUE(agent_name)
);
```

**字段说明**：

| 字段 | 类型 | 约束 | 说明 |
|------|------|------|------|
| `id` | INTEGER | PRIMARY KEY AUTOINCREMENT | 自增主键 |
| `agent_name` | TEXT | NOT NULL, UNIQUE | 智能体名称（如 `novel-master`） |
| `created_at` | TEXT | NOT NULL | 收藏时间（ISO 8601 格式） |

**索引**：
```sql
CREATE UNIQUE INDEX IF NOT EXISTS idx_agent_name ON agent_favorites(agent_name);
```

### 3.2 写入规范

遵循项目写入规范，使用 UPDATE + INSERT 两段式模式：

```python
# 收藏智能体
def add_favorite(agent_name: str) -> bool:
    now = datetime.now(UTC).isoformat()
    conn = get_connection()
    cursor = conn.cursor()
    
    # 先尝试更新（虽然这里只需要插入，但遵循规范）
    cursor.execute("""
        UPDATE agent_favorites
        SET created_at = ?
        WHERE agent_name = ?
    """, (now, agent_name))
    
    if cursor.rowcount == 0:
        cursor.execute("""
            INSERT INTO agent_favorites (agent_name, created_at)
            VALUES (?, ?)
        """, (agent_name, now))
    
    conn.commit()
    return True
```

### 4.1 新增文件

#### 4.1.1 收藏 API Hooks

位置：`frontend/src/core/agent-favorites/hooks.ts`

```typescript
import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
import { getBackendBaseURL } from "@/core/config";

// 获取收藏列表
export function useAgentFavorites() {
  return useQuery<string[]>({
    queryKey: ["agent-favorites"],
    queryFn: async () => {
      const response = await fetch(`${getBackendBaseURL()}/api/agent-favorites`);
      if (!response.ok) throw new Error("Failed to fetch favorites");
      const data = await response.json();
      return data.favorites;
    },
  });
}

// 添加收藏
export function useAddFavorite() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async (agentName: string) => {
      const response = await fetch(
        `${getBackendBaseURL()}/api/agent-favorites/${agentName}`,
        { method: "PUT" }
      );
      if (!response.ok) throw new Error("Failed to add favorite");
      return response.json();
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["agent-favorites"] });
    },
  });
}

// 取消收藏
export function useRemoveFavorite() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async (agentName: string) => {
      const response = await fetch(
        `${getBackendBaseURL()}/api/agent-favorites/${agentName}`,
        { method: "DELETE" }
      );
      if (!response.ok) throw new Error("Failed to remove favorite");
      return response.json();
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["agent-favorites"] });
    },
  });
}
```

#### 4.1.2 收藏按钮组件

位置：`frontend/src/components/workspace/agents/favorite-button.tsx`

```typescript
"use client";

import { HeartIcon } from "lucide-react";
import { Button } from "@/components/ui/button";
import { useAddFavorite, useRemoveFavorite } from "@/core/agent-favorites/hooks";

interface FavoriteButtonProps {
  agentName: string;
  isFavorited: boolean;
}

export function FavoriteButton({ agentName, isFavorited }: FavoriteButtonProps) {
  const addFavorite = useAddFavorite();
  const removeFavorite = useRemoveFavorite();

  const handleClick = (e: React.MouseEvent) => {
    e.stopPropagation();
    if (isFavorited) {
      removeFavorite.mutate(agentName);
    } else {
      addFavorite.mutate(agentName);
    }
  };

  return (
    <Button
      size="icon"
      variant="ghost"
      className="h-8 w-8 shrink-0"
      onClick={handleClick}
      title={isFavorited ? "取消收藏" : "收藏"}
    >
      <HeartIcon
        className={`h-4 w-4 ${
          isFavorited ? "fill-red-500 text-red-500" : "text-muted-foreground"
        }`}
      />
    </Button>
  );
}
```

## 5. 后端技术方案

### 5.1 新增 API 端点

文件：`backend/app/gateway/routers/agent_favorites.py`

```python
from fastapi import APIRouter, HTTPException
from datetime import datetime, UTC
import sqlite3
import logging

from deerflow.global_variables.storage import get_global_variables_db_path

logger = logging.getLogger(__name__)
router = APIRouter()


def _get_connection() -> sqlite3.Connection:
    db_path = get_global_variables_db_path()
    conn = sqlite3.connect(str(db_path))
    conn.row_factory = sqlite3.Row
    return conn


def _init_table():
    conn = _get_connection()
    conn.execute("""
        CREATE TABLE IF NOT EXISTS agent_favorites (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            agent_name TEXT NOT NULL,
            created_at TEXT NOT NULL,
            UNIQUE(agent_name)
        )
    """)
    conn.execute("""
        CREATE UNIQUE INDEX IF NOT EXISTS idx_agent_name 
        ON agent_favorites(agent_name)
    """)
    conn.commit()
    conn.close()


# 初始化表
_init_table()


@router.get("/agent-favorites")
async def list_favorites():
    """获取所有收藏的智能体列表"""
    try:
        conn = _get_connection()
        cursor = conn.execute(
            "SELECT agent_name FROM agent_favorites ORDER BY created_at DESC"
        )
        favorites = [row["agent_name"] for row in cursor.fetchall()]
        conn.close()
        return {"favorites": favorites}
    except Exception as e:
        logger.error(f"Failed to list favorites: {e}")
        raise HTTPException(status_code=500, detail=str(e))


@router.put("/agent-favorites/{agent_name}")
async def add_favorite(agent_name: str):
    """收藏智能体"""
    try:
        conn = _get_connection()
        now = datetime.now(UTC).isoformat()
        
        # 使用 UPDATE + INSERT 两段式
        conn.execute(
            "UPDATE agent_favorites SET created_at = ? WHERE agent_name = ?",
            (now, agent_name)
        )
        if conn.execute("SELECT changes()").fetchone()[0] == 0:
            conn.execute(
                "INSERT INTO agent_favorites (agent_name, created_at) VALUES (?, ?)",
                (agent_name, now)
            )
        
        conn.commit()
        conn.close()
        return {"success": True, "agent_name": agent_name}
    except Exception as e:
        logger.error(f"Failed to add favorite: {e}")
        raise HTTPException(status_code=500, detail=str(e))


@router.delete("/agent-favorites/{agent_name}")
async def remove_favorite(agent_name: str):
    """取消收藏智能体"""
    try:
        conn = _get_connection()
        conn.execute(
            "DELETE FROM agent_favorites WHERE agent_name = ?",
            (agent_name,)
        )
        conn.commit()
        conn.close()
        return {"success": True, "agent_name": agent_name}
    except Exception as e:
        logger.error(f"Failed to remove favorite: {e}")
        raise HTTPException(status_code=500, detail=str(e))
```

## 6. 业务流程

### 6.1 收藏智能体流程

```
用户在智能体列表页面浏览智能体
    ↓
点击智能体卡片上的收藏按钮（心形图标）
    ↓
前端调用 PUT /api/agent-favorites/{agent_name}
    ↓
后端将收藏记录写入 agent_favorites 表
    ↓
前端刷新收藏列表缓存
    ↓
左侧菜单实时更新显示收藏的智能体
```

### 6.2 取消收藏流程

```
用户点击已收藏智能体的收藏按钮（红色心形）
    ↓
前端调用 DELETE /api/agent-favorites/{agent_name}
    ↓
后端从 agent_favorites 表删除记录
    ↓
前端刷新收藏列表缓存
    ↓
左侧菜单实时更新，移除该智能体
```

### 6.3 快速访问流程

```
用户在左侧菜单看到收藏的智能体列表
    ↓
点击某个收藏的智能体名称
    ↓
前端跳转到 /workspace/agents/{agent_name}/chats/new
    ↓
进入与该智能体的新对话页面
```

## 9. 注意事项

1. **数据库表创建**：需要在用户允许的情况下创建新表 `agent_favorites`
2. **索引创建**：需要在用户允许的情况下创建唯一索引
3. **实时更新**：使用 React Query 的 `invalidateQueries` 实现收藏状态实时同步
4. **国际化**：收藏按钮的 tooltip 需要支持 i18n
5. **空状态处理**：左侧菜单无收藏时显示友好提示
6. **性能优化**：收藏列表请求可添加适当的缓存策略
